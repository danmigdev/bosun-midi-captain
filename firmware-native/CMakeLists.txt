cmake_minimum_required(VERSION 3.20)
set(BOSUN_PLATFORM host CACHE STRING "Build platform: host or rp2040")
set_property(CACHE BOSUN_PLATFORM PROPERTY STRINGS host rp2040)
if(BOSUN_PLATFORM STREQUAL "rp2040")
    include(cmake/pico_sdk.cmake)
    project(BosunNative VERSION 0.1.0 LANGUAGES C CXX ASM)
    pico_sdk_init()
elseif(BOSUN_PLATFORM STREQUAL "host")
    project(BosunNative VERSION 0.1.0 LANGUAGES C)
else()
    message(FATAL_ERROR "Unsupported BOSUN_PLATFORM=${BOSUN_PLATFORM}")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(BOSUN_SANITIZERS "Enable host address/undefined-behavior sanitizers" OFF)
add_library(bosun_checks INTERFACE)
target_compile_options(bosun_checks INTERFACE -Wall -Wextra -Werror)
if(BOSUN_SANITIZERS)
    if(NOT BOSUN_PLATFORM STREQUAL "host" OR NOT CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        message(FATAL_ERROR "BOSUN_SANITIZERS requires the host GNU/Clang build")
    endif()
    target_compile_options(bosun_checks INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(bosun_checks INTERFACE -fsanitize=address,undefined)
endif()

add_library(bosun_core STATIC src/json.c src/midi.c src/kemper.c src/switch_fsm.c)
target_include_directories(bosun_core PUBLIC include)
target_link_libraries(bosun_core PUBLIC bosun_checks m)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(manifest_header "${CMAKE_CURRENT_BINARY_DIR}/generated/bosun_manifest.h")
add_custom_command(OUTPUT "${manifest_header}"
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/cmake/build_manifest.py"
        --output "${manifest_header}"
    DEPENDS cmake/build_manifest.py src/runtime.c ../tools/build_manifest_tail.py
        ../firmware/lib/captain/messages.py ../firmware/lib/captain/plugin.py
        ../firmware/lib/plugins/kemper.py ../firmware/lib/plugins/generic_midi.py
        ../firmware/lib/plugins/headrush_core.py ../firmware/lib/plugins/ampero.py
        ../firmware/lib/plugins/line6_helix.py
    VERBATIM)
target_sources(bosun_core PRIVATE "${manifest_header}")
target_include_directories(bosun_core PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
# Required inventory: a missing component must fail CI, never silently build a
# partial application. No source glob can pull host fakes into the ARM image.
target_sources(bosun_core PRIVATE src/config.c src/runtime.c src/protocol.c
    src/application.c src/storage_path.c src/display.c src/expression_presence.c)

if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/littlefs/lfs.c")
    add_library(bosun_littlefs STATIC third_party/littlefs/lfs.c third_party/littlefs/lfs_util.c)
    target_include_directories(bosun_littlefs PUBLIC third_party/littlefs)
    target_compile_definitions(bosun_littlefs PUBLIC LFS_NO_MALLOC LFS_NO_DEBUG LFS_NO_WARN LFS_NO_ERROR LFS_NAME_MAX=63 LFS_FILE_MAX=262144)
    target_link_libraries(bosun_littlefs PUBLIC bosun_checks)
    # Upstream's assertion-only helper is intentionally unused with NDEBUG.
    target_compile_options(bosun_littlefs PRIVATE -Wno-unused-function)
endif()

if(BOSUN_PLATFORM STREQUAL "rp2040")
    add_subdirectory(platform/rp2040)
    target_compile_definitions(bosun_core PUBLIC BOSUN_PLATFORM_RP2040=1)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/platform/rp2040/storage.c")
        target_sources(bosun_core PRIVATE platform/rp2040/storage.c)
        target_link_libraries(bosun_core PUBLIC bosun_littlefs bosun_board)
    endif()
    add_executable(bosun_native src/main.c)
    target_link_libraries(bosun_native PRIVATE bosun_core)
    bosun_configure_rp2040_target(bosun_native)
else()
    enable_testing()
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    # Offline provisioning uses the real RP2040 littlefs backend with host NOR
    # memory. Never link the ordinary host-directory storage implementation.
    add_executable(bosun_storage_image platform/host/storage_image.c
        platform/rp2040/storage.c src/storage_path.c src/config.c src/json.c)
    target_include_directories(bosun_storage_image PRIVATE include)
    target_compile_definitions(bosun_storage_image PRIVATE BOSUN_PLATFORM_RP2040=1)
    target_link_libraries(bosun_storage_image PRIVATE bosun_checks bosun_littlefs)
    add_test(NAME storage_image COMMAND "${Python3_EXECUTABLE}"
        "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_storage_image.py"
        --builder "$<TARGET_FILE:bosun_storage_image>")
    set_tests_properties(storage_image PROPERTIES TIMEOUT 120)
    add_test(NAME hardware_benchmark_unit COMMAND "${Python3_EXECUTABLE}"
        "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_live_hardware_benchmark.py")
    set_tests_properties(hardware_benchmark_unit PROPERTIES TIMEOUT 60)
    add_test(NAME native_persistence_unit COMMAND "${Python3_EXECUTABLE}"
        "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_live_native_persistence.py")
    set_tests_properties(native_persistence_unit PROPERTIES TIMEOUT 60)
    add_test(NAME image_audit COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_image_audit.py")
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/platform/host/storage.c")
        target_sources(bosun_core PRIVATE platform/host/storage.c)
    endif()
    add_executable(bosun_emulator platform/host/main.c)
    target_link_libraries(bosun_emulator PRIVATE bosun_core)
    add_test(NAME host_emulator COMMAND "${Python3_EXECUTABLE}"
        "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_host_emulator.py"
        --emulator "$<TARGET_FILE:bosun_emulator>")
    set_tests_properties(host_emulator PROPERTIES TIMEOUT 120)
    function(bosun_add_test suite)
        add_executable(test_${suite} ${ARGN})
        target_link_libraries(test_${suite} PRIVATE bosun_checks)
        target_compile_options(test_${suite} PRIVATE -UNDEBUG)
        add_test(NAME ${suite} COMMAND test_${suite})
        set_tests_properties(${suite} PROPERTIES TIMEOUT 60)
    endfunction()
    bosun_add_test(board_contracts tests/test_board_contracts.c)
    target_include_directories(test_board_contracts PRIVATE include platform/rp2040)
    bosun_add_test(uart_tx tests/test_uart_tx.c)
    target_include_directories(test_uart_tx PRIVATE platform/rp2040)
    if(NOT PICO_SDK_PATH AND DEFINED ENV{PICO_SDK_PATH})
        set(PICO_SDK_PATH "$ENV{PICO_SDK_PATH}" CACHE PATH "Pico SDK for real descriptor host tests")
    endif()
    if(EXISTS "${PICO_SDK_PATH}/lib/tinyusb/src/tusb.h")
        bosun_add_test(usb_descriptors tests/test_usb_descriptors.c platform/rp2040/usb_descriptors.c)
        target_include_directories(test_usb_descriptors PRIVATE tests/fakes platform/rp2040 "${PICO_SDK_PATH}/lib/tinyusb/src")
        bosun_add_test(cdc_session tests/test_cdc_session.c platform/rp2040/cdc_session.c)
        target_include_directories(test_cdc_session PRIVATE include platform/rp2040 "${PICO_SDK_PATH}/lib/tinyusb/src")
    else()
        message(STATUS "USB descriptor host test needs PICO_SDK_PATH (all other host tests are SDK independent)")
    endif()
    foreach(suite json midi switch_fsm kemper config runtime protocol application storage display expression_presence)
        bosun_add_test(${suite} tests/test_${suite}.c)
        target_link_libraries(test_${suite} PRIVATE bosun_core)
    endforeach()
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_storage_lfs.c")
        bosun_add_test(storage_lfs tests/test_storage_lfs.c platform/rp2040/storage.c src/storage_path.c)
        target_include_directories(test_storage_lfs PRIVATE include)
        target_compile_definitions(test_storage_lfs PRIVATE BOSUN_PLATFORM_RP2040=1)
        target_link_libraries(test_storage_lfs PRIVATE bosun_littlefs)
    endif()
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/native_differential.c")
        add_executable(native_differential tests/native_differential.c)
        target_link_libraries(native_differential PRIVATE bosun_core)
        target_compile_options(native_differential PRIVATE -UNDEBUG)
        add_test(NAME differential COMMAND "${Python3_EXECUTABLE}"
            "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_differential.py"
            --driver "$<TARGET_FILE:native_differential>")
        set_tests_properties(differential PROPERTIES TIMEOUT 120)
    endif()
endif()
